cmake_minimum_required(VERSION 3.21)

project(NKR_ZKP LANGUAGES CXX)

set(NKR_ZKP_GROTH16_BACKEND "NONE" CACHE STRING
    "Groth16 accelerator to build: NONE, CPU, CUDA, or DCU")
set_property(CACHE NKR_ZKP_GROTH16_BACKEND PROPERTY STRINGS NONE CPU CUDA DCU)

string(TOUPPER "${NKR_ZKP_GROTH16_BACKEND}" NKR_ZKP_GROTH16_BACKEND)

if(NOT NKR_ZKP_GROTH16_BACKEND STREQUAL "NONE")
    add_subdirectory(protocols/groth16)
endif()
